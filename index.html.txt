    <script>
        // Estado global
        let selectedPlan = null;
        let selectedPlanName = '';
        let paymentLink = '';
        let selectedDate = null;
        let selectedTime = null;
        let currentMonth = new Date().getMonth();
        let currentYear = new Date().getFullYear();
        
        // Horários disponíveis (você pode editar aqui)
        const availableSlots = [
            '08:00', '08:30', '09:00', '09:30', '10:00', '10:30',
            '11:00', '11:30'
        ];
        
        // Horários ocupados (você atualiza manualmente aqui)
        // Formato: { 'YYYY-MM-DD': ['08:00', '09:00'] }
        const occupiedSlots = {
            // Exemplo: '2026-05-20': ['08:00', '09:00', '10:00']
        };
        
        // STEP 1: Selecionar Plano
        function selectPlan(price, name, link) {
            // Remove seleção anterior
            document.querySelectorAll('.plan-option').forEach(el => el.classList.remove('selected'));
            
            // Adiciona seleção
            document.getElementById(`plan-${price}`).classList.add('selected');
            
            // Atualiza estado
            selectedPlan = price;
            selectedPlanName = name;
            paymentLink = link;
            
            // Habilita botão
            document.getElementById('btn-next-step1').disabled = false;
        }
        
        // Navegação entre steps
        function goToStep(step) {
            // Esconde todos
            for(let i = 1; i <= 4; i++) {
                document.getElementById(`step${i}`).style.display = 'none';
                document.getElementById(`step-dot-${i}`).classList.remove('active', 'completed');
            }
            
            // Mostra step atual
            document.getElementById(`step${step}`).style.display = 'block';
            document.getElementById(`step-dot-${step}`).classList.add('active');
            
            // Marca anteriores como completos
            for(let i = 1; i < step; i++) {
                document.getElementById(`step-dot-${i}`).classList.add('completed');
            }
            
            // Ações específicas
            if(step === 2) {
                renderCalendar();
            }
            if(step === 3) {
                renderTimeSlots();
                document.getElementById('selected-date-display').textContent = formatDateBR(selectedDate);
            }
            if(step === 4) {
                updateSummary();
            }
            
            // Scroll to top
            window.scrollTo(0, 0);
        }
        
        // STEP 2: Calendário
        function renderCalendar() {
            const monthNames = ['Janeiro', 'Fevereiro', 'Março', 'Abril', 'Maio', 'Junho',
                               'Julho', 'Agosto', 'Setembro', 'Outubro', 'Novembro', 'Dezembro'];
            
            document.getElementById('calendar-month').textContent = 
                `${monthNames[currentMonth]} ${currentYear}`;
            
            const grid = document.getElementById('calendar-grid');
            grid.innerHTML = '';
            
            // Headers
            const days = ['Dom', 'Seg', 'Ter', 'Qua', 'Qui', 'Sex', 'Sáb'];
            days.forEach(day => {
                const header = document.createElement('div');
                header.className = 'calendar-day-header';
                header.textContent = day;
                grid.appendChild(header);
            });
            
            // Dias
            const firstDay = new Date(currentYear, currentMonth, 1).getDay();
            const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
            const today = new Date();
            
            // Espaços vazios antes do primeiro dia
            for(let i = 0; i < firstDay; i++) {
                const empty = document.createElement('div');
                grid.appendChild(empty);
            }
            
            // Dias do mês
            for(let day = 1; day <= daysInMonth; day++) {
                const date = new Date(currentYear, currentMonth, day);
                const dayEl = document.createElement('div');
                dayEl.className = 'calendar-day';
                dayEl.textContent = day;
                
                // Desabilita dias passados
                if(date < today) {
                    dayEl.classList.add('disabled');
                } else {
                    dayEl.onclick = () => selectDate(date);
                    
                    // Marca hoje
                    if(date.toDateString() === today.toDateString()) {
                        dayEl.classList.add('today');
                    }
                }
                
                grid.appendChild(dayEl);
            }
        }
        
        function previousMonth() {
            currentMonth--;
            if(currentMonth < 0) {
                currentMonth = 11;
                currentYear--;
            }
            renderCalendar();
        }
        
        function nextMonth() {
            currentMonth++;
            if(currentMonth > 11) {
                currentMonth = 0;
                currentYear++;
            }
            renderCalendar();
        }
        
        function selectDate(date) {
            document.querySelectorAll('.calendar-day').forEach(el => el.classList.remove('selected'));
            event.target.classList.add('selected');
            selectedDate = date;
            document.getElementById('btn-next-step2').disabled = false;
        }
        
        // STEP 3: Horários
        function renderTimeSlots() {
            const grid = document.getElementById('time-grid');
            grid.innerHTML = '';
            
            const dateKey = formatDateKey(selectedDate);
            const occupied = occupiedSlots[dateKey] || [];
            
            availableSlots.forEach(time => {
                const slot = document.createElement('div');
                slot.className = 'time-slot';
                slot.textContent = time;
                
                if(occupied.includes(time)) {
                    slot.classList.add('occupied');
                    slot.innerHTML = time + '<br><small>Ocupado</small>';
                } else {
                    slot.onclick = () => selectTime(time);
                }
                
                grid.appendChild(slot);
            });
        }
        
        function selectTime(time) {
            document.querySelectorAll('.time-slot').forEach(el => el.classList.remove('selected'));
            event.target.classList.add('selected');
            selectedTime = time;
            document.getElementById('btn-next-step3').disabled = false;
        }
        
        // STEP 4: Resumo
        function updateSummary() {
            document.getElementById('summary-plan').textContent = selectedPlanName;
            document.getElementById('summary-price').textContent = `R$ ${selectedPlan}`;
            document.getElementById('summary-date').textContent = formatDateBR(selectedDate);
            document.getElementById('summary-time').textContent = selectedTime;
        }
        
        function goToPayment() {
            // Abre Mercado Pago
            window.open(paymentLink, '_blank');
            
            // Mostra mensagem
            alert('Após o pagamento, envie o comprovante no WhatsApp: (44) 9133-9060\n\nVamos confirmar seu agendamento!');
        }
        
        // Helpers
        function formatDateKey(date) {
            const year = date.getFullYear();
            const month = String(date.getMonth() + 1).padStart(2, '0');
            const day = String(date.getDate()).padStart(2, '0');
            return `${year}-${month}-${day}`;
        }
        
        function formatDateBR(date) {
            const day = String(date.getDate()).padStart(2, '0');
            const month = String(date.getMonth() + 1).padStart(2, '0');
            const year = date.getFullYear();
            return `${day}/${month}/${year}`;
        }
        
        // Inicializar
        window.onload = () => {
            renderCalendar();
        };
    </script>
